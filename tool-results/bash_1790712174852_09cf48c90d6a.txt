    <script>
        // --- Three.js Scene Setup ---
        const container = document.getElementById('canvas-container');
        const scene = new THREE.Scene();
        
        // Camera
        const camera = new THREE.PerspectiveCamera(45, window.innerWidth / window.innerHeight, 0.1, 1000);
        camera.position.set(0, 2, 6);

        // Renderer
        const renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
        renderer.setSize(window.innerWidth, window.innerHeight);
        renderer.setPixelRatio(window.devicePixelRatio);
        renderer.toneMapping = THREE.ACESFilmicToneMapping;
        renderer.toneMappingExposure = 1.0;
        container.appendChild(renderer.domElement);

        // Lighting
        const ambientLight = new THREE.AmbientLight(0xffffff, 0.6);
        scene.add(ambientLight);

        const dirLight = new THREE.DirectionalLight(0xffccaa, 1.2);
        dirLight.position.set(5, 10, 7);
        scene.add(dirLight);

        // Geometry (The "Monolith")
        // Using a composite shape to match the "Arch" look in the screenshot roughly
        const group = new THREE.Group();

        // Base
        const baseGeo = new THREE.CylinderGeometry(1.2, 1.4, 0.5, 32);
        const matStone = new THREE.MeshStandardMaterial({ 
            color: 0xf0e6d2, 
            roughness: 0.85, 
            metalness: 0.05,
            flatShading: false
        });
        const base = new THREE.Mesh(baseGeo, matStone);
        base.position.y = -1.25;
        group.add(base);

        // Main Body (Extruded shape approximation)
        const shape = new THREE.Shape();
        shape.moveTo(-0.8, -1);
        shape.lineTo(-0.8, 1.5);
        shape.lineTo(-0.3, 1.5);
        shape.lineTo(-0.3, -0.2); // Inner arch left
        shape.lineTo(0.3, -0.2);  // Inner arch right
        shape.lineTo(0.3, 1.5);
        shape.lineTo(0.8, 1.5);
        shape.lineTo(0.8, -1);
        shape.lineTo(0.8, -1);
        shape.lineTo(-0.8, -1);

        const extrudeSettings = { depth: 1.0, bevelEnabled: true, bevelThickness: 0.1, bevelSize: 0.1, bevelSegments: 8 };
        const geometry = new THREE.ExtrudeGeometry(shape, extrudeSettings);
        geometry.center();
        
        // Rotate to stand up
        const mesh = new THREE.Mesh(geometry, matStone);
        mesh.rotation.x = Math.PI / 2;
        mesh.position.y = 0.2;
        group.add(mesh);

        scene.add(group);

        // Dust Particles System (Summer Effect)
        let dustParticles;
        const particleCount = 150;
        const particlesGeom = new THREE.BufferGeometry();
        const posArray = new Float32Array(particleCount * 3);
        
        for(let i = 0; i < particleCount * 3; i++) {
            posArray[i] = (Math.random() - 0.5) * 10;
        }
        
        particlesGeom.setAttribute('position', new THREE.BufferAttribute(posArray, 3));
        const particlesMat = new THREE.PointsMaterial({
            size: 0.03,
            color: 0xffffff,
            transparent: true,
            opacity: 0.6,
            blending: THREE.AdditiveBlending
        });
        
        dustParticles = new THREE.Points(particlesGeom, particlesMat);
        scene.add(dustParticles);


        // --- Interaction Logic ---

        // Season Toggle
        function setSeason(season) {
            const btnS = document.getElementById('btn-summer');
            const btnW = document.getElementById('btn-winter');
            const footS = document.getElementById('foot-summer');
            const footW = document.getElementById('foot-winter');

            if (season === 'summer') {
                // Visuals
                dirLight.color.setHex(0xffccaa);
                dirLight.intensity = 1.2;
                ambientLight.intensity = 0.6;
                dustParticles.visible = true;
                renderer.setClearColor(0x1a0f0a, 0); // Warm dark bg hint

                // UI State
                btnS.className = "flex-1 py-1.5 rounded-md text-xs font-semibold bg-amber-600/90 text-white shadow-sm transition-all";
                btnW.className = "flex-1 py-1.5 rounded-md text-xs font-semibold text-gray-400 hover:text-white transition-all";
                footS.className = "px-4 py-1 rounded-full text-xs font-bold bg-amber-600 text-white shadow";
                footW.className = "px-4 py-1 rounded-full text-xs font-bold text-gray-400 hover:text-white";

            } else {
                // Winter
                dirLight.color.setHex(0xaaccff);
                dirLight.intensity = 0.8;
                ambientLight.intensity = 0.4;
                dustParticles.visible = false;
                renderer.setClearColor(0x0a0f1a, 0); // Cool dark bg

                // UI State
                btnW.className = "flex-1 py-1.5 rounded-md text-xs font-semibold bg-blue-600/90 text-white shadow-sm transition-all";
                btnS.className = "flex-1 py-1.5 rounded-md text-xs font-semibold text-gray-400 hover:text-white transition-all";
                footW.className = "px-4 py-1 rounded-full text-xs font-bold bg-blue-600 text-white shadow";
                footS.className = "px-4 py-1 rounded-full text-xs font-bold text-gray-400 hover:text-white";
            }
        }

        // Info Card Toggle
        function toggleInfoCard() {
            const card = document.getElementById('info-card');
            if(card.style.opacity === '0') {
                card.style.opacity = '1';
                card.style.transform = 'translateY(0)';
                card.style.pointerEvents = 'auto';
            } else {
                card.style.opacity = '0';
                card.style.transform = 'translateY(20px)';
                card.style.pointerEvents = 'none';
            }
        }

        // Camera Presets
        function setCamera(view) {
            // Simple animation logic could go here, using GSAP or similar, but we'll do instant for vanilla
            const buttons = document.querySelectorAll('.preset-btn');
            buttons.forEach(b => {
                b.classList.remove('bg-amber-600/80', 'text-white', 'shadow-md');
                b.classList.add('text-gray-300');
            });
            event.currentTarget.classList.remove('text-gray-300');
            event.currentTarget.classList.add('bg-amber-600/80', 'text-white', 'shadow-md');

            if(view === 'hero') { camera.position.set(0, 2, 6); camera.lookAt(0,0,0); }
            else if(view === 'front') { camera.position.set(0, 0, 5); camera.lookAt(0,0,0); }
            else if(view === 'side') { camera.position.set(5, 0, 0); camera.lookAt(0,0,0); }
            else if(view === 'top') { camera.position.set(0, 6, 0.1); camera.lookAt(0,0,0); } // Cenital
        }

        function toggleFullscreen() {
            if (!document.fullscreenElement) {
                document.documentElement.requestFullscreen();
            } else {
                if (document.exitFullscreen) {
                    document.exitFullscreen();
                }
            }
        }

        // Animation Loop
        function animate() {
            requestAnimationFrame(animate);
            
            // Gentle rotation
            group.rotation.y += 0.002;

            // Dust movement
            if(dustParticles.visible) {
                const positions = dustParticles.geometry.attributes.position.array;
                for(let i=0; i<particleCount; i++) {
                    positions[i*3+1] += 0.005; // Float up
                    positions[i*3] += Math.sin(Date.now() * 0.001 + i) * 0.002; // Drift
                    if(positions[i*3+1] > 5) positions[i*3+1] = -5; // Reset
                }
                dustParticles.geometry.attributes.position.needsUpdate = true;
            }

            renderer.render(scene, camera);
        }

        // Handle Resize
        window.addEventListener('resize', () => {
            camera.aspect = window.innerWidth / window.innerHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
        });

        // Initialize
        setSeason('summer'); // Default state
        animate();

    </script>
